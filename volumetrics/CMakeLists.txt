cmake_minimum_required(VERSION 3.26)
project(Volumetrics LANGUAGES CXX)

if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(VOLUMETRICS_BUILD_OPENGL "Build the OpenGL pass DLL (requires host glad target)" ${VOLUMETRICS_BUILD_RENDERER})
option(VOLUMETRICS_BUILD_BENCHMARK "Build the standalone CPU benchmark" ON)
option(VOLUMETRICS_PROFILE_SYMBOLS "Keep symbols and frame pointers in Release builds" ON)

if(NOT TARGET glm::glm)
    find_package(glm 1.0.1 CONFIG QUIET)
    if(NOT TARGET glm::glm)
        include(FetchContent)
        FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git
            GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
        FetchContent_MakeAvailable(glm)
    endif()
endif()

function(volumetrics_profile_target target)
    set_target_properties(${target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(VOLUMETRICS_PROFILE_SYMBOLS)
        if(MSVC)
            target_compile_options(${target} PRIVATE /Zi /Oy-)
            target_link_options(${target} PRIVATE /DEBUG:FULL /OPT:NOICF)
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${target} PRIVATE -g -fno-omit-frame-pointer)
        endif()
    endif()
endfunction()

# The ONLY compiled copy of the sequential algorithm, used by both clients.
add_library(VolumetricCPU SHARED src/Lighting.cpp)
add_library(Volumetrics::CPU ALIAS VolumetricCPU)
target_compile_features(VolumetricCPU PUBLIC cxx_std_20)
target_compile_definitions(VolumetricCPU PRIVATE VOLUMETRIC_CPU_EXPORTS)
target_include_directories(VolumetricCPU PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(VolumetricCPU PUBLIC glm::glm)
set_target_properties(VolumetricCPU PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
volumetrics_profile_target(VolumetricCPU)

if(VOLUMETRICS_BUILD_OPENGL)
    if(NOT TARGET glad)
        message(FATAL_ERROR "OpenGL module needs the host glad target; use VOLUMETRICS_BUILD_OPENGL=OFF for CPU-only builds")
    endif()
    set_target_properties(glad PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(VolumetricFog SHARED src/Pass.cpp)
    add_library(Volumetrics::Fog ALIAS VolumetricFog)
    target_compile_definitions(VolumetricFog PRIVATE VOLUMETRIC_FOG_EXPORTS)
    target_link_libraries(VolumetricFog PUBLIC VolumetricCPU PRIVATE glad)
    set_target_properties(VolumetricFog PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    volumetrics_profile_target(VolumetricFog)
endif()

if(VOLUMETRICS_BUILD_BENCHMARK)
    add_executable(VolumetricBenchmark benchmark/main.cpp benchmark/Workload.cpp)
    target_link_libraries(VolumetricBenchmark PRIVATE VolumetricCPU)
    volumetrics_profile_target(VolumetricBenchmark)
    # All platforms can load the shared core beside the executable.
    add_custom_command(TARGET VolumetricBenchmark POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:VolumetricCPU> $<TARGET_FILE_DIR:VolumetricBenchmark>)
endif()

if(BUILD_TESTING)
    add_executable(VolumetricCPUTests tests/CPU.cpp benchmark/Workload.cpp)
    target_link_libraries(VolumetricCPUTests PRIVATE VolumetricCPU)
    add_test(NAME volumetric_cpu COMMAND VolumetricCPUTests)
    if(TARGET VolumetricBenchmark)
        add_test(NAME benchmark_smoke COMMAND VolumetricBenchmark --iterations 2 --warmup 1 --width 40 --height 30 --steps 16 --shadow-size 32)
        add_test(NAME benchmark_invalid_input COMMAND VolumetricBenchmark --iterations 0)
        set_tests_properties(benchmark_invalid_input PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
